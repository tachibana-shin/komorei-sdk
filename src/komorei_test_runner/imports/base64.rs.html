9/// Encode raw bytes as standard (RFC 4648 §4) base64 with padding.
10pub fn encode(mut env: FunctionEnvMut<WasmEnv>, ptr: Ptr, len: u32) -> FFIResult {
11	let Ok(data) = env.data().read_bytes(&env, ptr, len) else {
12		return -1;
13	};
14	env.data_mut().store.store(StoreItem::String(STANDARD.encode(&data)))
15}
16
17/// Tolerant decode: standard padded/unpadded and URL-safe padded/unpadded.
18pub fn decode(mut env: FunctionEnvMut<WasmEnv>, ptr: Ptr, len: u32) -> FFIResult {
19	let Ok(data) = env.data().read_string(&env, ptr, len) else {
20		return -1;
21	};
22	let decoded = STANDARD
23		.decode(&data)
24		.or_else(|_| STANDARD_NO_PAD.decode(&data))
25		.or_else(|_| URL_SAFE.decode(&data))
26		.or_else(|_| URL_SAFE_NO_PAD.decode(&data));
27	match decoded {
28		Ok(bytes) => env.data_mut().store.store(StoreItem::Encoded(bytes)),
29		Err(_) => -1,
30	}
31}
