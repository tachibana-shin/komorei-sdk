1//! Module for base64 encoding and decoding.
2//!
3//! Implemented natively by the runner (Rust `base64` crate) — sources never
4//! have to carry their own decoder, which keeps the wasm smaller and the
5//! decode faster. The runner tolerates all four common variants (standard
6//! padded/unpadded and URL-safe padded/unpadded), matching what real grab
7//! pages ship.
8use super::FFIResult;
9use crate::alloc::{String, Vec};
10use crate::imports::std::{destroy, read_buffer, read_string_and_destroy};
11
12#[link(wasm_import_module = "base64")]
13unsafe extern "C" {
14	#[link_name = "encode"]
15	fn _encode(ptr: *const u8, len: usize) -> FFIResult;
16
17	#[link_name = "decode"]
18	fn _decode(ptr: *const u8, len: usize) -> FFIResult;
19}
20
21/// Encode raw bytes as standard (RFC 4648 §4) base64 with padding.
22pub fn encode(data: &[u8]) -> String {
23	let rid = unsafe { _encode(data.as_ptr(), data.len()) };
24	// Base64 output is ASCII; the empty result maps to the empty string too.
25	read_string_and_destroy(rid).unwrap_or_default()
26}
27
28/// Decode a base64 string (any tolerated variant) into raw bytes.
29///
30/// Returns `None` when the input is not valid base64.
31pub fn decode(data: &str) -> Option<Vec<u8>> {
32	let rid = unsafe { _decode(data.as_ptr(), data.len()) };
33	if rid < 0 {
34		return None;
35	}
36	let buffer = read_buffer(rid);
37	unsafe { destroy(rid) };
38	buffer
39}
