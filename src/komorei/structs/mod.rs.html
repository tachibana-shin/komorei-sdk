46/// The preferred update strategy for an anime.
47///
48/// Titles marked as `Always` will be included in library refreshes by default,
49/// while `Never` will be excluded. Useful for titles that are known to be fully
50/// completed or have a static episode list that won't change after the initial fetch.
51#[derive(Default, PartialEq, Eq, Clone, Copy, Debug, Serialize, Deserialize)]
52pub enum UpdateStrategy {
53	#[default]
54	Always,
55	Never,
56}

69/// A specific season or part of a franchise.
70///

73pub struct AnimeSeason {
74	/// The anime id of the season.

76	/// Title of the season.

93/// An anime, series, or other type of content for Komorei to watch.
94///
95/// Usually fetched in two stages: Lite (listing) and Full (details + episodes).
96#[derive(Default, Clone, Debug, PartialEq, Serialize, Deserialize)]
97pub struct Anime {
98	/// Unique identifier for the anime.
99	pub key: String,
100	/// Identifier of the source this anime belongs to.
101	pub source_id: String,
102	/// Title of the anime.
103	pub title: String,
104	/// Original (raw) title of the anime.
105	pub original_title: String,
106	/// Link to the anime cover (poster) image.
107	pub cover: String,
108	/// Link to the anime banner image.
109	pub banner: Option<String>,
110	/// Description of the anime.
111	pub description: Option<String>,
112	/// Total number of episodes.
113	pub episode_count: i32,
114	/// Current episode label, e.g. "Tập 12/12" or "HD Vietsub".
115	pub current_episode: Option<String>,
116	/// Rating of the anime (0 to 10).
117	pub rating: Option<f32>,
118	/// Number of ratings.
119	pub rating_count: Option<i32>,
120	/// Publishing status of the anime.
121	pub status: AnimeStatus,
122	/// Year the anime was released.
123	pub release_year: Option<CategoryLink>,
124	/// List of genres.
125	pub genres: Vec<CategoryLink>,
126	/// List of authors.
127	pub authors: Vec<CategoryLink>,
128	/// Studio that produced the anime.
129	pub studio: Option<CategoryLink>,
130	/// The season this anime belongs to.
131	pub season_of: Option<CategoryLink>,
132	/// List of countries.
133	pub countries: Vec<CategoryLink>,
134	/// Whether the anime is featured (e.g. on the home banner).
135	pub is_featured: bool,
136	/// Number of views.
137	pub views: i32,
138	/// Info on the next episode airing, e.g. "Tập 13 phát sóng 20:00 thứ 7".
139	pub next_episode_air_info: Option<String>,
140	/// Quality tag of the anime, e.g. "FHD".
141	pub quality_tag: Option<String>,
142	/// List of seasons.
143	pub seasons: Vec<AnimeSeason>,
144	/// List of episodes. Only populated when `needs_chapters` is set.
145	pub episodes: Option<Vec<Episode>>,
146	/// Link to the anime on the source website.
147	pub url: Option<String>,
148}
149
150impl Anime {
151	/// Copy the values from another anime into this one.
152	///
153	/// Used to upgrade a Lite anime with full details and episodes.
154	pub fn copy_from(&mut self, anime: Anime) {
155		self.key = anime.key;
156		self.source_id = anime.source_id;
157		self.title = anime.title;
158		self.original_title = anime.original_title;
159		self.cover = anime.cover;
160		if let Some(banner) = anime.banner {
161			self.banner = Some(banner);
162		}
163		if let Some(description) = anime.description {
164			self.description = Some(description);
165		}
166		if anime.episode_count != 0 {
167			self.episode_count = anime.episode_count;
168		}
169		if let Some(current_episode) = anime.current_episode {
170			self.current_episode = Some(current_episode);
171		}
172		if let Some(rating) = anime.rating {
173			self.rating = Some(rating);
174		}
175		if let Some(rating_count) = anime.rating_count {
176			self.rating_count = Some(rating_count);
177		}
178		self.status = anime.status;
179		if let Some(release_year) = anime.release_year {
180			self.release_year = Some(release_year);
181		}
182		if !anime.genres.is_empty() {
183			self.genres = anime.genres;
184		}
185		if !anime.authors.is_empty() {
186			self.authors = anime.authors;
187		}
188		if let Some(studio) = anime.studio {
189			self.studio = Some(studio);
190		}
191		if let Some(season_of) = anime.season_of {
192			self.season_of = Some(season_of);
193		}
194		if !anime.countries.is_empty() {
195			self.countries = anime.countries;
196		}
197		if anime.views != 0 {
198			self.views = anime.views;
199		}
200		if let Some(next_episode_air_info) = anime.next_episode_air_info {
201			self.next_episode_air_info = Some(next_episode_air_info);
202		}
203		if let Some(quality_tag) = anime.quality_tag {
204			self.quality_tag = Some(quality_tag);
205		}
206		if !anime.seasons.is_empty() {
207			self.seasons = anime.seasons;
208		}
209		if let Some(episodes) = anime.episodes {
210			self.episodes = Some(episodes);
211		}
212		if let Some(url) = anime.url {
213			self.url = Some(url);
214		}
215	}
216}

227/// An episode of an anime.
228#[derive(Default, Clone, Debug, PartialEq, Serialize, Deserialize)]
229pub struct Episode {

231	pub key: String,
232	/// Episode number, as displayed by the source (e.g. "1", "12.5", "HD").
233	pub episode_number: String,
234	/// Title of the episode (excluding the episode number).
235	pub title: Option<String>,
236	/// Date the episode was uploaded (unix timestamp).
237	pub date_uploaded: Option<i64>,
238	/// Optional thumbnail image url for the episode.
239	pub thumbnail: Option<String>,
240	/// Quality of the episode, e.g. "1080p FHD".
241	pub quality: Option<String>,
242	/// Duration of the episode in seconds (reference only; real duration comes from the stream).
243	pub duration_seconds: Option<i64>,
244	/// Link to the episode on the source website.
245	pub url: Option<String>,
246	/// Language of the episode.
247	pub language: Option<String>,
248	/// Boolean indicating if the episode is locked.
249	pub locked: bool,
250}
