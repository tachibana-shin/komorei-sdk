1//! Streamable video data structures shared between Komorei and sources.
2
3use super::HashMap;
4use crate::alloc::{String, Vec};
5use serde::{Deserialize, Serialize};
6
7/// The streaming format of a media source.
8#[derive(Default, PartialEq, Eq, Clone, Copy, Debug, Serialize, Deserialize)]
9pub enum StreamType {
10	/// HTTP Live Streaming (m3u8 playlist).
11	#[default]
12	HLS,
13	/// Direct MP4 file.
14	MP4,
15	/// DASH (mpd manifest).
16	DASH,
17	/// Any other resolvable format.
18	OTHER,
19}
20
21/// A playable stream (server) for an episode.
22#[derive(Default, Clone, Debug, PartialEq, Serialize, Deserialize)]
23pub struct StreamInfo {
24	/// Unique identifier for the stream.
25	pub key: String,
26	/// Display name of the stream/server.
27	pub name: String,
28	/// Quality label of the stream.
29	pub quality: String,
30}
31
32/// A millisecond time range — used to mark an episode's opening (intro) or
33/// ending (outro) segment on the progress bar and for skip buttons.
34#[derive(Default, Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
35pub struct RangeLong {
36	/// Start of the range, in milliseconds.
37	pub start_ms: i64,
38	/// End of the range, in milliseconds.
39	pub end_ms: i64,
40}
41
42/// Resolved stream data used by the media engine.
43#[derive(Default, Clone, Debug, PartialEq, Serialize, Deserialize)]
44pub struct StreamData {
45	/// The media url, or the raw media content itself when `is_content` is true.
46	pub url: String,
47	/// The streaming format.
48	pub stream_type: StreamType,
49	/// Whether `url` holds the raw media content itself — e.g. an HLS playlist
50	/// text starting with "#EXTM3U...". `false` means the url must be resolved
51	/// (fetched) before playback.
52	pub is_content: bool,
53	/// Extra headers (Referer, User-Agent, ...) applied to every media sub-request.
54	pub headers: HashMap<String, String>,
55	/// Optional subtitle tracks.
56	pub subtitles: Vec<SubtitleInfo>,
57	/// Time range of the opening (intro) segment.
58	pub intro: Option<RangeLong>,
59	/// Time range of the ending (outro) segment.
60	pub outro: Option<RangeLong>,
61}
62
63/// A subtitle track for a stream.
64#[derive(Default, Clone, Debug, PartialEq, Serialize, Deserialize)]
65pub struct SubtitleInfo {
66	/// Url of the subtitle file.
67	pub url: String,
68	/// Language of the subtitle (e.g. "vi").
69	pub language: String,
70	/// Display label of the subtitle.
71	pub label: Option<String>,
72	/// Extra headers for fetching this subtitle.
73	pub headers: HashMap<String, String>,
74}
