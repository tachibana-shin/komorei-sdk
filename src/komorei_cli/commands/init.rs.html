41#[allow(clippy::too_many_arguments)]
42pub fn run(
43	path: Option<std::path::PathBuf>,
44	name: Option<String>,
45	url: Option<String>,
46	mut languages: Vec<String>,
47	content_rating: Option<SourceContentRating>,
48	template: bool,
49	template_name: Option<String>,
50	sdk_path: Option<std::path::PathBuf>,
51) -> anyhow::Result<()> {
52	// verify already input languages
53	for lang in &languages {
54		if !VALID_LANGUAGE_CODES.contains(&lang.as_str()) {
55			return Err(anyhow!("Invalid language code: {}", lang));
56		}
57	}
58
59	let current_dir = std::env::current_dir().context("Failed to get current directory")?;
60	let target_path = path.unwrap_or(current_dir.clone());
61
62	// use directory name as default source name
63	let directory_name = target_path
64		.file_name()
65		.context("Failed to get directory name")?
66		.to_string_lossy();
67
68	let template_name = if template {
69		Some(match template_name {
70			Some(n) => n,
71			None => Input::<String>::new()
72				.with_prompt("Template name")
73				.interact_text()?,
74		})
75	} else {
76		None
77	};
78
79	// if name is None, prompt for it. use directory name as default
80	let name = match name {
81		Some(n) => n,
82		None => Input::new()
83			.with_prompt("Source name")
84			.default(directory_name.to_string())
85			.interact_text()?,
86	};
87
88	// if url is None, prompt for it
89	let url = match url {
90		Some(u) => u,
91		None => Input::new()
92			.with_prompt("Source URL")
93			.validate_with(|input: &String| -> Result<(), &str> {
94				if input.starts_with("http://") || input.starts_with("https://") {
95					Ok(())
96				} else {
97					Err("URL must start with http:// or https://")
98				}
99			})
100			.interact_text()?,
101	};
102
103	// get language codes
104	if languages.is_empty() {
105		let available_languages = [
106			("English", "en"),
107			("Chinese", "zh"),
108			("Spanish", "es"),
109			("Russian", "ru"),
110			("Vietnamese", "vi"),
111			("Other", ""),
112		];
113
114		let language_names: Vec<&str> = available_languages.iter().map(|(name, _)| *name).collect();
115
116		let selections = MultiSelect::new()
117			.with_prompt("Select supported languages")
118			.items(&language_names)
119			.interact()?;
120
121		if selections.is_empty() {
122			return Err(anyhow!("You must select at least one language"));
123		}
124
125		for i in selections {
126			if language_names[i] == "Other" {
127				let other_input: String = Input::new()
128					.with_prompt("Enter additional language codes (e.g. `id pt ja`)")
129					.validate_with(|input: &String| -> Result<(), &str> {
130						let codes = input.split_whitespace().map(|s| s.to_lowercase());
131						for code in codes {
132							if !VALID_LANGUAGE_CODES.contains(&code.as_str()) {
133								return Err(
134									"One or more language codes are not valid ISO 639-2 codes",
135								);
136							}
137						}
138						Ok(())
139					})
140					.interact_text()?;
141
142				let mut other_codes: Vec<String> = other_input
143					.split_whitespace()
144					.map(|s| s.to_lowercase())
145					.collect();
146
147				languages.append(&mut other_codes);
148			} else {
149				let code = available_languages[i].1.to_string();
150				languages.push(code);
151			}
152		}
153	}
154
155	// get content rating
156	let content_rating = match content_rating {
157		Some(rating) => rating,
158		None => {
159			let rating_options = &["Safe", "Contains NSFW content", "Primarily NSFW content"];
160			let selection = Select::new()
161				.with_prompt("Select content rating")
162				.items(rating_options)
163				.default(0)
164				.interact()?;
165
166			match selection {
167				0 => SourceContentRating::Safe,
168				1 => SourceContentRating::ContainsNsfw,
169				2 => SourceContentRating::PrimarilyNsfw,
170				_ => unreachable!(),
171			}
172		}
173	};
174
175	// create and move to the target directory if necessary
176	if target_path != current_dir {
177		std::fs::create_dir_all(&target_path).context("Failed to create new directory")?;
178		std::env::set_current_dir(&target_path).context("Failed to change directory")?;
179	}
180
181	// locate the komorei-sdk checkout used for dependency resolution. Prefer an
182	// explicit `--komorei` flag, then auto-detect a `komorei-sdk/` directory
183	// among `target_path`'s ancestors (the standard layout keeps the SDK as a
184	// separate git checkout next to the source collection). Falling back to the
185	// git dependency requires the SDK to be published.
186	let sdk_root: Option<std::path::PathBuf> = match sdk_path {
187		Some(p) => Some(std::path::absolute(&p).unwrap_or(p)),
188		// the current directory is the (already entered) target directory
189		None => {
190			find_komorei_sdk(&std::env::current_dir().context("Failed to get current directory")?)
191		}
192	};
193	match &sdk_root {
194		Some(root) => println!("Using komorei-sdk at {}", root.display()),
195		None => println!(
196			"warning: komorei-sdk not found near {}, falling back to the git dependency",
197			target_path.display()
198		),
199	}
200
201	// determine info from configuration
202	fn create_package_name(name: &str, template: bool) -> String {
203		let name = name
204			.chars()
205			.map(|c| c.to_ascii_lowercase()) // lowercase
206			.map(|c| if c == ' ' { '-' } else { c }) // replace spaces with hyphens
207			.filter(|c| c.is_ascii_alphanumeric() || *c == '-' || *c == '_') // ensure unicode-xid compliant
208			.collect::<String>()
209			.trim_matches(|c| c == '_' || c == '-') // remove trailing and leading underscores and hyphens
210			.to_string();
211		if name == "test" {
212			if template {
213				"test_template".to_string()
214			} else {
215				"test_source".to_string()
216			}
217		} else {
218			name
219		}
220	}
221	let mut package_name = create_package_name(&name, false);
222	// handle edge case for reserved package names
223	if package_name == "test" {
224		package_name = "test_source".to_string();
225	}
226	let id = format!(
227		"{}.{}",
228		if languages.len() == 1 {
229			&languages[0]
230		} else {
231			"multi"
232		},
233		package_name
234	);
235	let source_json = SourceJson {
236		info: SourceInfo {
237			id,
238			name,
239			alt_names: None,
240			version: 1,
241			url: Some(url),
242			urls: None,
243			content_rating: Some(content_rating),
244			languages,
245			min_app_version: None,
246			max_app_version: None,
247		},
248	};
249
250	if let Some(template_name) = template_name {
251		let template_package_name = create_package_name(&template_name, false);
252		let template_name = template_name
253			.chars()
254			.filter(|c| c.is_ascii_alphanumeric())
255			.collect::<String>();
256
257		// create workspace Cargo.toml
258		let cargo_toml = PathBuf::from("Cargo.toml");
259		std::fs::write(&cargo_toml, CARGO_TEMPLATE)
260			.context("Failed to create workspace Cargo.toml")?;
261
262		create_template_files(&template_package_name, &template_name, sdk_root.as_deref())?;
263		create_source_files(
264			&package_name,
265			source_json,
266			Some((&template_package_name, &template_name)),
267			sdk_root.as_deref(),
268		)?;
269	} else {
270		create_source_files(&package_name, source_json, None, sdk_root.as_deref())?;
271		println!("configured a source");
272	}
273
274	Ok(())
275}

277// create a new cargo library and add komorei dependencies to it
278// moves the current directory into a newly created directory if in_workspace is true
279fn create_empty_komorei_lib(
280	name: &str,
281	in_workspace: bool,
282	template_name: Option<(&str, &str)>,
283	sdk_root: Option<&std::path::Path>,
284) -> Result<(), anyhow::Error> {
285	// initialize via cargo
286	let mut cmd = &mut std::process::Command::new("cargo");
287	if in_workspace {
288		// if we're in a workspace, create a new directory
289		cmd = cmd.arg("new").arg(name);
290	} else {
291		// otherwise, use the current directory
292		cmd = cmd.arg("init").arg("--name").arg(name);
293	}
294	cmd.arg("--lib")
295		.stdout(std::process::Stdio::null())
296		.stderr(std::process::Stdio::null())
297		.status()
298		.context("Failed to initialize via cargo")?;
299
300	// if this is a workspace, move into the new directory we created
301	if in_workspace {
302		std::env::set_current_dir(PathBuf::from(name)).context("Failed to change directory")?;
303	}
304
305	// add komorei library dependencies
306	let cargo_dir = std::env::current_dir().context("Failed to get current directory")?;
307
308	if let Some(sdk_root) = sdk_root {
309		// local development: reference the komorei-sdk checkout by relative path
310		// (the SDK is not published, so a git dependency can't be resolved yet)
311		let lib_dir = relative_path(&cargo_dir, &sdk_root.join("crates").join("lib"));
312		let test_macro_dir = relative_path(&cargo_dir, &sdk_root.join("crates").join("test-macro"));
313
314		cargo_add_path_dependency("komorei", &lib_dir, &["json"], false)?;
315		cargo_add_path_dependency("komorei", &lib_dir, &["test"], true)?;
316		cargo_add_path_dependency("komorei-test", &test_macro_dir, &[], true)?;
317	} else {
318		// fall back to the (as yet unpublished) git repository
319		cargo_add_git_dependency("komorei", &["json"], false)?;
320		cargo_add_git_dependency("komorei", &["test"], true)?;
321		cargo_add_git_dependency("komorei-test", &[], true)?;
322	}
323
324	// if a template name is provided, add it as a dependency
325	if let Some((template_package_name, _)) = template_name {
326		let mut add_template_cmd = std::process::Command::new("cargo");
327		let result = add_template_cmd
328			.arg("add")
329			.arg("--path")
330			.arg(format!("../{template_package_name}"))
331			.stdout(std::process::Stdio::null())
332			.stderr(std::process::Stdio::null())
333			.status()
334			.context("Failed to initialize source via cargo")?;
335		if !result.success() {
336			return Err(anyhow!("Failed to initialize source via cargo"));
337		}
338	}
339
340	Ok(())
341}

343// add a path-based dependency to the crate in the current directory
344fn cargo_add_path_dependency(
345	package: &str,
346	path: &std::path::Path,
347	features: &[&str],
348	dev: bool,
349) -> Result<(), anyhow::Error> {
350	let mut cmd = std::process::Command::new("cargo");
351	cmd.arg("add").arg(package).arg("--path").arg(path);
352	for feature in features {
353		cmd.arg("--features").arg(feature);
354	}
355	if dev {
356		cmd.arg("--dev");
357	}
358	let result = cmd
359		.stdout(std::process::Stdio::null())
360		.stderr(std::process::Stdio::null())
361		.status()
362		.context("Failed to initialize source via cargo")?;
363	if !result.success() {
364		return Err(anyhow!("Failed to add {} dependency", package));
365	}
366	Ok(())
367}

369// add a git-based dependency to the crate in the current directory
370fn cargo_add_git_dependency(
371	package: &str,
372	features: &[&str],
373	dev: bool,
374) -> Result<(), anyhow::Error> {
375	let mut cmd = std::process::Command::new("cargo");
376	cmd.arg("add")
377		.arg(package)
378		.arg("--git")
379		.arg("https://github.com/komorei-sdk/komorei-sdk.git");
380	for feature in features {
381		cmd.arg("--features").arg(feature);
382	}
383	if dev {
384		cmd.arg("--dev");
385	}
386	let result = cmd
387		.stdout(std::process::Stdio::null())
388		.stderr(std::process::Stdio::null())
389		.status()
390		.context("Failed to initialize source via cargo")?;
391	if !result.success() {
392		return Err(anyhow!("Failed to add {} dependency", package));
393	}
394	Ok(())
395}

397// locate a `komorei-sdk` directory among `from`'s ancestors
398fn find_komorei_sdk(from: &std::path::Path) -> Option<std::path::PathBuf> {
399	let mut current = Some(from.to_path_buf());
400	while let Some(dir) = current {
401		if dir.join("komorei-sdk").join("Cargo.toml").is_file() {
402			return Some(dir.join("komorei-sdk"));
403		}
404		current = dir.parent().map(|p| p.to_path_buf());
405	}
406	None
407}

409// compute a relative path from `from` to `to` (used for cargo path dependencies)
410fn relative_path(from: &std::path::Path, to: &std::path::Path) -> std::path::PathBuf {
411	let from_abs = std::path::absolute(from).unwrap_or_else(|_| from.to_path_buf());
412	let to_abs = std::path::absolute(to).unwrap_or_else(|_| to.to_path_buf());
413
414	let from_components: Vec<_> = from_abs.components().collect();
415	let to_components: Vec<_> = to_abs.components().collect();
416
417	let common = from_components
418		.iter()
419		.zip(&to_components)
420		.take_while(|(a, b)| a == b)
421		.count();
422
423	let mut out = std::path::PathBuf::new();
424	for _ in common..from_components.len() {
425		out.push("..");
426	}
427	for component in &to_components[common..] {
428		out.push(component.as_os_str());
429	}
430	out
431}

433// creates files for a new source
434fn create_source_files(
435	package_name: &str,
436	source_json: SourceJson,
437	template_name: Option<(&str, &str)>,
438	sdk_root: Option<&std::path::Path>,
439) -> Result<(), anyhow::Error> {
440	let in_workspace = template_name.is_some();
441
442	// creates and moves into our new project folder if necessary
443	create_empty_komorei_lib(package_name, in_workspace, template_name, sdk_root)?;
444
445	// override lib.rs
446	let lib_rs_path = PathBuf::from("src/lib.rs");
447	let source_name = source_json
448		.info
449		.name
450		.chars()
451		.filter(|c| c.is_ascii_alphanumeric())
452		.collect::<String>();
453	let lib_rs_content = if let Some((template_package_name, template_name)) = template_name {
454		SOURCE_TEMPLATE_LIB_TEMPLATE
455			.replace("{{SOURCE_NAME}}", &source_name)
456			.replace("{{TEMPLATE_NAME}}", template_name)
457			.replace(
458				"{{TEMPLATE_LIB_NAME}}",
459				&template_package_name.replace("-", "_"),
460			)
461	} else {
462		SOURCE_LIB_TEMPLATE.replace("{{SOURCE_NAME}}", &source_name)
463	};
464	std::fs::write(&lib_rs_path, lib_rs_content).context("Failed to write template lib.rs")?;
465
466	// add wasm config stuff on to Cargo.toml
467	let cargo_toml = PathBuf::from("Cargo.toml");
468	let mut cargo_toml_content =
469		std::fs::read_to_string(&cargo_toml).context("Failed to read Cargo.toml")?;
470	cargo_toml_content.push_str(CDYLIB_STR);
471	// if we're not in a workspace, add the profile stuff
472	if !in_workspace {
473		// use the profile configuration from workspace Cargo.toml template
474		let profile_text = CARGO_TEMPLATE
475			.lines()
476			.skip(3)
477			.collect::<Vec<_>>()
478			.join("\n");
479		cargo_toml_content.push_str(&profile_text);
480	}
481	std::fs::write(&cargo_toml, cargo_toml_content).context("Failed to write Cargo.toml")?;
482
483	// create .cargo/config.toml
484	let config_path = PathBuf::from(".cargo");
485	std::fs::create_dir_all(&config_path).context("Failed to create .cargo directory")?;
486	let config_toml = config_path.join("config.toml");
487	std::fs::write(&config_toml, CONFIG_TEMPLATE).context("Failed to create config.toml")?;
488
489	// create res folder
490	let res_path = PathBuf::from("res");
491	std::fs::create_dir_all(&res_path).context("Failed to create res directory")?;
492
493	// create source.json
494	let source_json_path = res_path.join("source.json");
495	let tabs_pretty = serde_json::ser::PrettyFormatter::with_indent(b"\t"); // indent with tabs
496	let file = std::fs::File::create(&source_json_path).context("Failed to create json file")?;
497	let mut w = std::io::BufWriter::new(file);
498	let mut ser = serde_json::Serializer::with_formatter(&mut w, tabs_pretty);
499	serde::Serialize::serialize(&source_json, &mut ser).context("Failed to write to json file")?;
500	w.write_all(b"\n")?; // add a newline after the json data
501	w.flush()?;
502
503	// create empty icon.png
504	let icon_path = res_path.join("icon.png");
505	std::fs::File::create(&icon_path).context("Failed to create empty icon.png")?;
506
507	// if we created and moved into a directory, move out of it
508	if in_workspace {
509		std::env::set_current_dir(PathBuf::from("..")).context("Failed to change directory")?;
510	}
511
512	Ok(())
513}

515// creates files for a new source template
516fn create_template_files(
517	package_name: &str,
518	name: &str,
519	sdk_root: Option<&std::path::Path>,
520) -> Result<(), anyhow::Error> {
521	create_empty_komorei_lib(package_name, true, None, sdk_root)?;
522
523	// override lib.rs
524	let lib_rs_path = PathBuf::from("src/lib.rs");
525	let lib_rs_content = TEMPLATE_LIB_TEMPLATE.replace("{{TEMPLATE_NAME}}", name);
526	std::fs::write(&lib_rs_path, lib_rs_content).context("Failed to write lib.rs template")?;
527
528	// move out of the directory we created
529	std::env::set_current_dir(PathBuf::from("..")).context("Failed to change directory")?;
530
531	Ok(())
532}
